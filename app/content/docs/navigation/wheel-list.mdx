## Usage

```tsx
import { 
  WheelList,
  WheelOption,
} from "@/components/ui/pxl/wheel-list"
```

```tsx
const items = [
  { label: "Light", value: "light" },
  { label: "Dark", value: "dark" },
  { label: "System", value: "system" },
];

<WheelList>
  {items.map((item) => (
    <WheelOption key={item.value} value={item.value}>
      {item.label}
    </WheelOption>
  ))}
</WheelList>
```

## Composition

Use the following composition to build a `WheelList`:

```text
WheelList
└── WheelOption
```
