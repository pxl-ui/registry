## Usage

```tsx
import { 
  SectionHeading,
  SectionHeadingTagline,
  SectionHeadingTitle,
  SectionHeadingBody,
} from "@/components/features/pxl/marketing/section-heading";
```

```tsx
<SectionHeading>
  <SectionHeadingTagline>
    Pricing
  </SectionHeadingTagline>
  <SectionHeadingTitle>
    Simple Pricing, Smarter Work
  </SectionHeadingTitle>
  <SectionHeadingBody>
    Acme Inc. has a plan designed to help you move faster.
  </SectionHeadingBody>
</SectionHeading>
```
