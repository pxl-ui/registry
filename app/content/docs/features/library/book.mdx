## Usage

```tsx
import {
  Book,
  BookBackCover,
  BookCover,
  BookSpine,
} from "@/components/features/pxl/library/book"
```

```tsx
<Book view="cover" size="lg">
  <BookBackCover>Back cover content</BookBackCover>
  <BookSpine>Spine content</BookSpine>
  <BookCover>Cover content</BookCover>
</Book>
```

## API Reference

### Book

| Prop | Default | Description |
| --- | --- | --- |
| `view` | `"spine"` | `"cover"`, `"spine"`, or `"back-cover"`. |
| `size` | `"sm"` | `"sm"`, `"md"` or `"lg"`. |
| `duration` | `0.65` | Turn duration in seconds. |
