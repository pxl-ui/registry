## Usage

```tsx
import {
  ProfileCard,
  ProfileCardAvatar,
  ProfileCardBody,
  ProfileCardDetails,
  ProfileCardName,
  ProfileCardVerifiedBadge,
} from "@/components/ui/pxl/profile-card"
```

```tsx
<ProfileCard>
  <ProfileCardAvatar
    name="@hero"
    src="https://raw.githubusercontent.com/pxl-ui/registry/main/app/public/portraits/hero.png" 
  />
  <ProfileCardDetails>
    <ProfileCardName>
      Link <ProfileCardVerifiedBadge />
    </ProfileCardName>
    <ProfileCardBody>@hero</ProfileCardBody>
  </ProfileCardDetails>
</ProfileCard>
```
## Composition

Use the following composition to build a `ProfileCard`

```text
ProfileCard
├── ProfileCardAvatar
└── ProfileCardDetails
    ├── ProfileCardName
    │   └──ProfileCardVerifiedBadge
    └── ProfileCardBody
```
