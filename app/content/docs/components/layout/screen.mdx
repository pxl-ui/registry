## Usage

```tsx

  ScreenFilter, 
  ScreenProvider
} from "@/components/ui/pxl/screen";
```

```tsx
<ScreenProvider variant="lcd">
  <ScreenFilter>
  </ScreenFilter>
</ScreenProvider>
```

## Composition

Use the following composition to build a `Screen`:

```text
ScreenProvider
└── ScreenFilter
```
